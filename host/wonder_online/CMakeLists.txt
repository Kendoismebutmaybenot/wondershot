cmake_minimum_required(VERSION 3.20)
project(wonder_online LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(GEN_FILES OFF CACHE BOOL "" FORCE)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_definitions(MBEDTLS_SSL_DTLS_SRTP)
add_compile_options("-ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=online" "-ffile-prefix-map=${WONDER_DEPS}=dependencies")
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(USE_SHARED_MBEDTLS_LIBRARY OFF CACHE BOOL "" FORCE)
set(DISABLE_PACKAGE_CONFIG_AND_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory(${WONDER_DEPS}/mbedtls mbedtls)
add_library(MbedTLS::MbedTLS ALIAS mbedtls)
add_library(MbedTLS::MbedX509 ALIAS mbedx509)
add_library(MbedTLS::MbedCrypto ALIAS mbedcrypto)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(USE_MBEDTLS ON CACHE BOOL "" FORCE)
set(NO_MEDIA ON CACHE BOOL "" FORCE)
set(NO_WEBSOCKET ON CACHE BOOL "" FORCE)
set(NO_TESTS ON CACHE BOOL "" FORCE)
set(NO_EXAMPLES ON CACHE BOOL "" FORCE)
add_subdirectory(${WONDER_DEPS}/libdatachannel datachannel)
if(MINGW)
  # usrsctp's Windows C code uses the Windows min/max macros.
  target_compile_options(usrsctp PRIVATE -UNOMINMAX)
endif()
add_library(wonder_online STATIC transport.cpp identity.cpp)
target_include_directories(wonder_online PRIVATE ${WONDER_DEPS}/libdatachannel/deps/json/include ${WONDER_DEPS}/mbedtls/include)
target_link_libraries(wonder_online PUBLIC datachannel-static winhttp)
